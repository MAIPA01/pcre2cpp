1699 class basic_regex {

1702
1703 using _code_type = typename _pcre2_data_t::code_type;
1704 using _code_ptr = std::shared_ptr<_code_type>;

1708 using _string_view_type = typename _pcre2_data_t::string_view_type;
1709 using _string_char_type = typename _pcre2_data_t::string_char_type;
1710 using _match_value_type = basic_match_value<utf>;
1711 using _match_result_type = basic_match_result<utf>;
1712 using _sptr_type = typename _pcre2_data_t::sptr_type;
1713 using _named_sub_values_table = typename _pcre2_data_t::named_sub_values_table;
1714 using _named_sub_values_table_ptr = std::shared_ptr<_named_sub_values_table>;
1715 using _uchar_type = typename _pcre2_data_t::uchar_type;
1716 #if _PCRE2CPP_HAS_EXCEPTIONS
1717 using _regex_exception = basic_regex_exception<utf>;
1718 #endif
1719
1720 /// @brief pointer to compiled pcre2 code
1721 _code_ptr _code = nullptr;
1722 /// @brief pointer to match data of pcre2 code
1723 _match_data_ptr _match_data = nullptr;
1724 /// @brief pointer to conversion table of named groups to their index
1725 _named_sub_values_table_ptr _named_sub_values = nullptr;
1726
1727 public:
1728 /// @brief basic regex container with pattern and compile options
1729 _PCRE2CPP_CONSTEXPR20 explicit basic_regex(const _string_view_type pattern,
1730 const compile_options opts = compile_options_bits::None) _PCRE2CPP_NOEXCEPT {
1731 // Compile Code
1732 int error_code;
1733 size_t error_offset;
1734
1735 _code_type* code = _pcre2_data_t::compile(reinterpret_cast<_sptr_type>(pattern.data()), pattern.size(), opts,
1736 &error_code, &error_offset, nullptr);
1737
1738
1739 #if !_PCRE2CPP_HAS_EXCEPTIONS
1740 pcre2cpp_assert(code != nullptr, "Failed to initialize code: {}",
1741 convert_any_utf_to_utf8<utf>(generate_error_message<utf>(error_code, error_offset)));
1742 #else
1743 if (code == nullptr) { throw _regex_exception(error_code, error_offset); }
1744 #endif
1745
1746
1747 _code = std::shared_ptr<_code_type>(code, _pcre2_data_t::code_free);
1748
1749 // Get Named Sub Values
1750 _named_sub_values = std::make_shared<_named_sub_values_table>();
1751
1752 size_t name_count = 0;
1753 _uchar_type* name_table = nullptr;
1754 size_t name_entry_size = 0;
1755
1756 _pcre2_data_t::get_info(_code.get(), PCRE2_INFO_NAMECOUNT, &name_count);
1757 _pcre2_data_t::get_info(_code.get(), PCRE2_INFO_NAMETABLE, &name_table);
1758 _pcre2_data_t::get_info(_code.get(), PCRE2_INFO_NAMEENTRYSIZE, &name_entry_size);
1759
1760 for (size_t i = 0; i != name_count; ++i) {
1761 _uchar_type* entry = name_table + i * name_entry_size + 2;
1762 const int index = _pcre2_data_t::substring_number_from_name(_code.get(), entry);
1763
1764 _uchar_type* entry_end = entry + 1;
1765 while (*entry_end != 0 && static_cast<size_t>(entry_end - entry) < name_entry_size - 3) { entry_end += 1; }
1766 _named_sub_values->emplace(_string_view_type(reinterpret_cast<_string_char_type*>(entry), entry_end - entry),
1767 static_cast<size_t>(index) - 1);
1768 }
1769
1770 // Create Match Data
1771 _match_data_type* match_data = _pcre2_data_t::match_data_from_pattern(_code.get(), nullptr);
1772 _match_data = std::shared_ptr<_match_data_type>(match_data, _pcre2_data_t::match_data_free);
1773 }
